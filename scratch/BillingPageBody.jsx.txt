                                    <tbody className="divide-y-0">
                                        {filteredBills.map((bill) => {
                                            const normalizedBillType = resolveBillType(bill.billType);

                                                <tr key={bill._id} className="group bg-slate-50/50 hover:bg-blue-50/30 transition-all duration-300">

                                                            <div className={`w-10 h-10 rounded-xl flex items-center justify-center shadow-sm ${normalizedBillType === 'SALES' ? 'bg-emerald-50 text-emerald-600' : 'bg-blue-50 text-blue-600'}`}>
                                                                <FileText size={16} />
                                                            </div>

                                                            <p className="text-[10px] font-bold text-slate-400 tracking-wider mt-0.5">{bill.customer?.phone}</p>

                                                                bill.paymentStatus === 'paid' ? 'bg-emerald-100 text-emerald-700' :

                                                                bill.paymentStatus === 'partial' ? 'bg-blue-100 text-blue-700' : 
                                                                'bg-rose-100 text-rose-700'

                                                            <option value="cancelled">Cancelled</option>
                                                        </select>
                                                    </td>
                                                    <td className="px-6 py-5 rounded-r-3xl border-y border-transparent group-hover:border-blue-100 text-right">
                                                        <div className="flex items-center justify-end gap-2 opacity-0 group-hover:opacity-100 transition-opacity">
                                                            <button className="w-8 h-8 rounded-lg bg-blue-50 text-blue-600 hover:bg-blue-600 hover:text-white transition-all flex items-center justify-center shadow-sm" onClick={() => handleViewBill(bill)} title="Audit Details"><Eye size={14} /></button>

                                                            <button className="w-8 h-8 rounded-lg bg-purple-50 text-purple-600 hover:bg-purple-600 hover:text-white transition-all flex items-center justify-center shadow-sm" onClick={() => { setEmailBill(bill); setEmailTo(pickDefaultRecipient(bill)); setShowEmailModal(true); }} title="Email"><Mail size={14} /></button>
                                                            <button className="w-8 h-8 rounded-lg bg-rose-50 text-rose-600 hover:bg-rose-600 hover:text-white transition-all flex items-center justify-center shadow-sm" onClick={() => handleDeleteClick(bill)} title="Delete"><Trash2 size={14} /></button>
                                                        </div>
                                                    </td>
                                                </tr>
                                            );
                                        })}
                                    </tbody>
                                </table>
                            </div>

                    </div>

                    {/* Modals Container */}
                    <div className="modals-container">
                        {/* Preview Modal */}

                                        <div className="flex items-center gap-4">

                                                <p className="text-[10px] font-bold text-slate-400 uppercase tracking-widest">{selectedBill.billNumber}</p>

                                    <div className="p-8 bg-slate-100/50 flex justify-center max-h-[70vh] overflow-y-auto">

                                <div className="modal-content max-w-lg" onClick={(e) => e.stopPropagation()}>
                                    <div className="modal-header">
                                        <div className="flex items-center gap-3">
                                            <div className="w-10 h-10 rounded-xl bg-purple-50 flex items-center justify-center text-purple-600"><Mail size={20} /></div>
                                            <div>
                                                <h3 className="text-xl font-black text-slate-900 tracking-tight">Dispatch Invoice</h3>
                                                <p className="text-[10px] font-bold text-slate-400 uppercase tracking-widest">{emailBill.billNumber}</p>
                                            </div>
                                        </div>
                                        <button className="action-btn hover:bg-slate-100" onClick={closeEmailModal}><X size={20} /></button>
                                    </div>
                                    <div className="p-8 space-y-6">
                                        <div>
                                            <label className="text-[10px] font-black text-slate-400 uppercase tracking-widest mb-3 block ml-1">Recipient Address *</label>
                                            <div className="relative">
                                                <AtSign size={18} className="absolute left-4 top-1/2 -translate-y-1/2 text-slate-400" />
                                                <input
                                                    type="email"

                                    </div>
                                </div>
                            </div>
                        )}

                        {/* Delete Modal */}
                        {showDeleteModal && deleteBill && (
                            <div className="modal-overlay" onClick={() => setShowDeleteModal(false)}>
                                <div className="modal-content max-w-md" onClick={(e) => e.stopPropagation()}>
                                    <div className="p-10 text-center">
                                        <div className="w-20 h-20 rounded-3xl bg-rose-50 text-rose-600 flex items-center justify-center mx-auto mb-8 animate-bounce shadow-sm border border-rose-100"><Trash2 size={32} /></div>
                                        <h3 className="text-2xl font-black text-slate-900 tracking-tight mb-3">Purge Transaction?</h3>
                                        <p className="text-slate-500 font-medium">This action will permanently erase <span className="font-black text-slate-900">{deleteBill.billNumber}</span> from the central ledger. This cannot be reversed.</p>
                                        <div className="grid grid-cols-2 gap-4 mt-10">
                                            <button className="btn btn-secondary py-4" onClick={() => setShowDeleteModal(false)}>Retain Record</button>
                                            <button className="btn btn-primary bg-rose-600 hover:bg-rose-700 py-4 shadow-lg shadow-rose-500/20" onClick={handleDeleteConfirm}>Delete Forever</button>

                        {/* Template Modal */}

                                <div className="modal-content max-w-[240mm]" onClick={(e) => e.stopPropagation()}>
                                    <div className="modal-header border-b border-slate-100 p-8">
                                        <div className="flex items-center gap-4">
                                            <div className="w-14 h-14 rounded-2xl bg-slate-900 text-white flex items-center justify-center shadow-lg"><FileText size={24} /></div>
                                            <div>
                                                <h3 className="text-2xl font-black text-slate-900 tracking-tight">Format Template</h3>
                                                <p className="text-[10px] font-bold text-slate-400 uppercase tracking-widest">Master Layout Reference</p>
                                            </div>
                                        </div>
                                        <div className="flex items-center gap-4">
                                            <button className="btn btn-primary" onClick={handleDownloadEmptyTemplate}>Download Template</button>
                                            <button className="action-btn hover:bg-slate-100" onClick={() => setShowEmptyInvoiceModal(false)}><X size={24} /></button>
                                        </div>
                                    </div>
                                    <div className="p-8 bg-slate-100/50 flex justify-center max-h-[75vh] overflow-y-auto">
                                        <div ref={emptyInvoiceRef} className="shadow-2xl bg-white">
                                            <BillTemplate
                                                bill={{
                                                    billNumber: 'TEMP-000',
                                                    date: new Date(),
                                                    customer: { name: '', phone: '', address: '', gstin: '', state: 'Tamilnadu', stateCode: '33' },
